Core Utility Functions

Daily Word, Prayer Engine, Recommendation Logic

lib/dailyWord.ts getDailyWord

export function getDailyWord(words: any[]) {
  const today = new Date().toISOString().slice(0, 10);
  const index =
    today.split("").reduce((sum, char) => sum + char.charCodeAt(0), 0) %
    words.length;

  return words[index];
}

lib/prayerEngine.ts buildPrayer

export function buildPrayer(words: any[]) {
  const meanings = words.map((w) => w.meaning).join(" -> ");

  return {
    sequence: words.map((w) => w.word),
    meaningFlow: meanings,
    prayer: `Father, lead me through ${meanings.toLowerCase()} according to Your promise.`
  };
}

lib/recommendations.ts recommendCluster

export function recommendCluster(userNeed: string) {
  const need = userNeed.toLowerCase();

  if (need.includes("peace")) return "Peace & Protection";
  if (need.includes("healing")) return "Healing & Restoration";
  if (need.includes("wisdom")) return "Light & Revelation";
  if (need.includes("purpose")) return "Calling & Purpose";
  if (need.includes("identity")) return "Sonship & Belonging";

  return "Calling & Purpose";
}
